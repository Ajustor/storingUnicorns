**Goal:** An egui desktop interface with feature parity with the TUI: connections, schema tree, SQL editor with tabs/highlighting/completion, virtualised results grid, row editing, table structure, import/export, truncate, in-app updates.

**Architecture:** `src/gui/` is an `eframe::App`. All database work goes through `gui::worker::Worker`, which owns a tokio runtime, spawns `engine::ops` calls and sends `Event`s back over a channel drained every frame; the UI thread never blocks. The connection lives in the app as `Arc<DatabaseConnection>` and is cloned into each spawned task. Pure logic (filtering, completion insertion, form validation, highlighting colours) lives in small functions with unit tests; rendering code is verified manually.

**Tech Stack:** eframe/egui 0.31, egui_extras 0.31 (TableBuilder), egui-phosphor 0.9 (icons), rfd 0.15 (native file dialogs), tokio.

    System,
    Dark,
    Light,

    pub fn from_config(value: Option<&str>) -> Self {

            Some("dark") => Self::Dark,
            Some("light") => Self::Light,
            _ => Self::System,
        }
    }

```rust
//! Runs database work off the UI thread. Each command is spawned on a tokio
//! runtime; its outcome comes back as an `Event`, drained by `poll()` every
//! frame. `repaint` wakes the UI when an event is ready.

use std::future::Future;
use std::path::PathBuf;
use std::sync::mpsc::{self, Receiver, Sender};
use std::sync::Arc;

use tokio::runtime::Runtime;
use tokio::task::JoinHandle;

use crate::engine::db::DatabaseConnection;
use crate::engine::models::{Column, ConnectionConfig, QueryResult, SchemaInfo};
use crate::engine::ops::{self, query::Executed, rows::BatchReport, transfer::ImportStats};
use crate::engine::services::export_import::{export_to_file, ExportFormat};
use crate::engine::services::table_cache::TableCache;
use crate::engine::services::{ColumnDefinition, SchemaModification};

pub type Conn = Arc<DatabaseConnection>;

/// Result of a background operation. Errors are carried as display strings.
pub enum Event {
    Connected { config: ConnectionConfig, conn: Conn, schemas: Vec<SchemaInfo> },
    ConnectFailed { name: String, error: String },
    TestFinished(Result<(), String>),
    Schemas(Result<Vec<SchemaInfo>, String>),
    QueryFinished(Result<Executed, String>),
    /// `row` is the index in the current result that was edited.
    RowUpdated { row: usize, values: Vec<String>, outcome: Result<u64, String> },
    RowInserted(Result<u64, String>),
    RowDeleted { row: usize, outcome: Result<u64, String> },
    Columns { table: String, outcome: Result<Vec<ColumnDefinition>, String> },
    SchemaApplied { table: String, outcome: Result<String, String> },
    Exported(Result<(usize, PathBuf), String>),
    Imported { table: String, outcome: Result<ImportStats, String> },
    /// `kind` is "Export", "Import" or "Vidage".
    Batch { kind: &'static str, report: BatchReport },
    Progress { done: usize, total: usize, label: String },
}

pub struct Worker {
    rt: Runtime,
    tx: Sender<Event>,
    rx: Receiver<Event>,
    repaint: Arc<dyn Fn() + Send + Sync>,
    /// The query that `cancel()` aborts.
    running: Option<JoinHandle<()>>,
    pub cache: Arc<TableCache>,
}

fn err(e: impl std::fmt::Display) -> String {
    e.to_string()
}

impl Worker {
    pub fn new(repaint: impl Fn() + Send + Sync + 'static) -> Self {
        let (tx, rx) = mpsc::channel();
        Self {
            rt: Runtime::new().expect("tokio runtime"),
            tx,
            rx,
            repaint: Arc::new(repaint),
            running: None,
            cache: Arc::new(TableCache::default()),
        }
    }

    pub fn poll(&mut self) -> Vec<Event> {
        self.rx.try_iter().collect()
    }

    pub fn is_running(&self) -> bool {
        self.running.as_ref().is_some_and(|h| !h.is_finished())
    }

    /// Abort the running query. Returns whether something was cancelled.
    /// Dropping a transaction's dedicated connection without COMMIT makes the
    /// database roll it back.
    pub fn cancel(&mut self) -> bool {
        match self.running.take() {
            Some(h) if !h.is_finished() => {
                h.abort();
                true
            }
            _ => false,
        }
    }

    fn spawn<F>(&self, fut: F) -> JoinHandle<()>
    where
        F: Future<Output = Event> + Send + 'static,
    {
        let tx = self.tx.clone();
        let repaint = self.repaint.clone();
        self.rt.spawn(async move {
            let _ = tx.send(fut.await);
            repaint();
        })
    }

    /// Progress callback usable from inside a spawned task.
    fn progress(&self) -> impl FnMut(usize, usize, &str) + Send + 'static {
        let tx = self.tx.clone();
        let repaint = self.repaint.clone();
        move |done, total, label| {
            let _ = tx.send(Event::Progress { done, total, label: label.to_string() });
            repaint();
        }
    }

    pub fn connect(&self, config: ConnectionConfig) {
        self.cache_clear();
        self.spawn(async move {
            match DatabaseConnection::connect(&config).await {
                Ok(conn) => {
                    let schemas = ops::query::refresh_schemas(&conn).await.unwrap_or_default();
                    Event::Connected { config, conn: Arc::new(conn), schemas }
                }
                Err(e) => Event::ConnectFailed { name: config.name, error: err(e) },
            }
        });
    }

    pub fn test_connection(&self, config: ConnectionConfig) {
        self.spawn(async move {
            let outcome = match DatabaseConnection::connect(&config).await {
                Ok(conn) => {
                    let r = conn.test().await.map_err(err);
                    conn.close().await;
                    r
                }
                Err(e) => Err(err(e)),
            };
            Event::TestFinished(outcome)
        });
    }

    fn cache_clear(&self) {
        let cache = self.cache.clone();
        self.rt.spawn(async move { cache.clear().await });
    }

    pub fn refresh_schemas(&self, conn: Conn) {
        self.spawn(async move { Event::Schemas(ops::query::refresh_schemas(&conn).await.map_err(err)) });
    }

    pub fn run_at_cursor(&mut self, conn: Conn, text: String, cursor: usize) {
        self.cancel();
        self.running = Some(self.spawn(async move {
            Event::QueryFinished(ops::query::run_at_cursor(&conn, &text, cursor).await.map_err(err))
        }));
    }

    pub fn run_all(&mut self, conn: Conn, text: String) {
        self.cancel();
        self.running = Some(self.spawn(async move {
            Event::QueryFinished(ops::query::run_all(&conn, &text).await.map(Executed::Query).map_err(err))
        }));
    }

    pub fn update_row(&self, conn: Conn, table: String, columns: Vec<Column>, row: usize, original: Vec<String>, values: Vec<String>) {
        self.spawn(async move {
            let outcome = ops::rows::update_row(&conn, &table, &columns, &original, &values).await.map_err(err);
            Event::RowUpdated { row, values, outcome }
        });
    }

    pub fn insert_row(&self, conn: Conn, table: String, columns: Vec<Column>, values: Vec<String>, system: Vec<usize>) {
        self.spawn(async move {
            Event::RowInserted(ops::rows::insert_row(&conn, &table, &columns, &values, &system).await.map_err(err))
        });
    }

    pub fn delete_row(&self, conn: Conn, table: String, columns: Vec<Column>, row: usize, values: Vec<String>, quotes: (char, char)) {
        self.spawn(async move {
            let outcome = ops::rows::delete_row(&conn, &table, &columns, &values, quotes).await.map_err(err);
            Event::RowDeleted { row, outcome }
        });
    }

    pub fn fetch_columns(&self, conn: Conn, table: String) {
        let cache = self.cache.clone();
        self.spawn(async move {
            let outcome = ops::schema::fetch_columns(&conn, &cache, &table).await.map_err(err);
            Event::Columns { table, outcome }
        });
    }

    pub fn apply_schema(&self, conn: Conn, m: SchemaModification, db_type: crate::engine::models::DatabaseType) {
        let cache = self.cache.clone();
        self.spawn(async move {
            let table = ops::schema::table_of(&m).unwrap_or_default().to_string();
            let outcome = ops::schema::apply_modification(&conn, &cache, &m, &db_type).await.map_err(err);
            Event::SchemaApplied { table, outcome }
        });
    }

    pub fn export_result(&self, result: QueryResult, format: ExportFormat, path: PathBuf, table: String, quotes: (char, char)) {
        self.spawn(async move {
            let outcome = tokio::task::spawn_blocking(move || {
                export_to_file(&result, format, &path.to_string_lossy(), &table, quotes.0, quotes.1)
                    .map(|n| (n, path))
            })
            .await
            .map_err(err)
            .and_then(|r| r);
            Event::Exported(outcome)
        });
    }

    pub fn import_csv(&self, conn: Conn, table: String, path: PathBuf, quotes: (char, char)) {
        let mut progress = self.progress();
        self.spawn(async move {
            let outcome = match tokio::fs::read_to_string(&path).await {
                Ok(content) => ops::transfer::import_csv(&conn, &table, &content, quotes, |d, t| progress(d, t, "lignes"))
                    .await
                    .map_err(err),
                Err(e) => Err(format!("{}: {e}", path.display())),
            };
            Event::Imported { table, outcome }
        });
    }

    pub fn export_tables(&self, conn: Conn, tables: Vec<(String, String)>, dir: PathBuf, format: ExportFormat, quotes: (char, char)) {
        let progress = self.progress();
        self.spawn(async move {
            let report = ops::transfer::export_tables(&conn, &tables, &dir, format, quotes, progress).await;
            Event::Batch { kind: "Export", report }
        });
    }

    pub fn import_tables(&self, conn: Conn, tables: Vec<(String, String)>, dir: PathBuf, quotes: (char, char)) {
        let progress = self.progress();
        self.spawn(async move {
            let report = ops::transfer::import_tables(&conn, &tables, &dir, quotes, progress).await;
            Event::Batch { kind: "Import", report }
        });
    }

    pub fn truncate_tables(&self, conn: Conn, tables: Vec<String>) {
        let progress = self.progress();
        self.spawn(async move {
            let report = ops::rows::truncate_tables(&conn, &tables, progress).await;
            Event::Batch { kind: "Vidage", report }
        });
    }
}
```

impl App {
    pub fn new(cc: &eframe::CreationContext<'_>, exit_action: Arc<Mutex<Option<ExitAction>>>) -> Self {
        let config = AppConfig::load().unwrap_or_default();
        let theme = ThemeChoice::from_config(config.theme.as_deref());
        cc.egui_ctx.set_fonts(theme::fonts());
        theme::apply(&cc.egui_ctx, theme);

        let ctx = cc.egui_ctx.clone();
        let worker = Worker::new(move || ctx.request_repaint());
        let ctx = cc.egui_ctx.clone();
        let mut updater = Updater::new(move || ctx.request_repaint());
        updater.check(false, config.skipped_version.as_deref());

        let mut tabs = QueryTabsState::load().unwrap_or_default();
        if tabs.tabs.is_empty() {
            tabs.add_tab();
        }
        tabs.active_tab = tabs.active_tab.min(tabs.tabs.len() - 1);
        let selected_connection = config
            .last_connection
            .as_ref()
            .and_then(|n| config.connections.iter().position(|c| &c.name == n));

        Self {
            config,
            theme,
            exit_action,
            worker,
            updater,
            session: None,
            connecting: None,
            selected_connection,
            schemas: Vec::new(),
            table_filter: String::new(),
            tabs,
            cursor: 0,
            known_columns: Vec::new(),
            completion: Default::default(),
            results: None,
            status: Status { text: "Prêt".into(), kind: StatusKind::Info },
            busy: None,
            progress: None,
            dialog: None,
        }
    }

    pub fn info(&mut self, text: impl Into<String>) {
        self.status = Status { text: text.into(), kind: StatusKind::Info };
    }
    pub fn success(&mut self, text: impl Into<String>) {
        self.status = Status { text: text.into(), kind: StatusKind::Success };
    }
    pub fn error(&mut self, text: impl Into<String>) {
        self.status = Status { text: text.into(), kind: StatusKind::Error };
    }

    pub fn conn(&self) -> Option<Conn> {
        self.session.as_ref().map(|s| s.conn.clone())
    }

    pub fn quotes(&self) -> (char, char) {
        self.session.as_ref().map(|s| s.quotes).unwrap_or(('"', '"'))
    }

    pub fn save_config(&mut self) {
        if let Err(e) = self.config.save() {
            self.error(format!("Impossible d'enregistrer la configuration : {e}"));
        }
    }

    pub fn connect(&mut self, index: usize) {
        let Some(config) = self.config.connections.get(index).cloned() else { return };
        self.connecting = Some(index);
        self.selected_connection = Some(index);
        self.busy = Some((format!("Connexion à {}…", config.name), Instant::now()));
        self.worker.connect(config);
    }

    pub fn run_current(&mut self) {
        let Some(conn) = self.conn() else {
            self.error("Pas de connexion active");
            return;
        };
        let text = self.tabs.current_tab().query.clone();
        self.busy = Some(("Exécution…".into(), Instant::now()));
        self.worker.run_at_cursor(conn, text, self.cursor);
    }

    pub fn run_all(&mut self) {
        let Some(conn) = self.conn() else {
            self.error("Pas de connexion active");
            return;
        };
        let text = self.tabs.current_tab().query.clone();
        self.busy = Some(("Exécution…".into(), Instant::now()));
        self.worker.run_all(conn, text);
    }

    pub fn cancel(&mut self) {
        if self.worker.cancel() {
            self.busy = None;
            self.info("Requête annulée");
        }
    }

    pub fn refresh_schemas(&mut self) {
        if let Some(conn) = self.conn() {
            self.worker.refresh_schemas(conn);
        }
    }

    pub fn show_result(&mut self, result: QueryResult, sql: &str) {
        self.known_columns = result.columns.iter().map(|c| c.name.clone()).collect();
        let visible = (0..result.rows.len()).collect();
        self.results = Some(ResultView {
            result,
            table: extract_table_from_query(sql),
            filter: String::new(),
            visible,
            selected: None,
        });
    }

    fn handle_event(&mut self, ev: Event) {
        match ev {
            Event::Connected { config, conn, schemas } => {
                self.busy = None;
                self.connecting = None;
                self.success(format!("Connecté à {}", config.name));
                self.config.last_connection = Some(config.name.clone());
                self.save_config();
                let quotes = quote_chars(&config.db_type);
                self.schemas = schemas;
                self.session = Some(Session { config, conn, quotes });
            }
            Event::ConnectFailed { name, error } => {
                self.busy = None;
                self.connecting = None;
                self.error(format!("Connexion à {name} impossible : {error}"));
            }
            Event::Schemas(Ok(schemas)) => {
                let n: usize = schemas.iter().map(|s| s.tables.len()).sum();
                self.schemas = schemas;
                self.info(format!("{n} tables"));
            }
            Event::Schemas(Err(e)) => self.error(format!("Lecture des tables impossible : {e}")),
            Event::QueryFinished(outcome) => {
                self.busy = None;
                let sql = self.tabs.current_tab().query.clone();
                match outcome {
                    Ok(Executed::Query(r)) => {
                        let msg = if r.columns.is_empty() {
                            format!("{} ligne(s) affectée(s) en {} ms", r.rows_affected, r.execution_time_ms)
                        } else {
                            format!("{} ligne(s) en {} ms", r.rows.len(), r.execution_time_ms)
                        };
                        self.show_result(r, &sql);
                        self.success(msg);
                    }
                    Ok(Executed::Transaction { result, statements }) => {
                        let msg = format!("Transaction validée : {statements} instructions en {} ms", result.execution_time_ms);
                        self.show_result(result, &sql);
                        self.success(msg);
                    }
                    Err(e) => self.error(e),
                }
            }
            Event::Progress { done, total, label } => self.progress = Some((done, total, label)),
            other => super::dialogs::handle_event(self, other),
        }
    }

    fn shortcuts(&mut self, ctx: &egui::Context) {
        if self.dialog.is_some() {
            return;
        }
        let pressed = |s: &KeyboardShortcut| ctx.input_mut(|i| i.consume_shortcut(s));
        if pressed(&RUN_CURRENT) {
            self.run_current();
        }
        if pressed(&RUN_ALL) {
            self.run_all();
        }
        if pressed(&NEW_TAB) {
            self.tabs.add_tab();
        }
        if pressed(&CLOSE_TAB) {
            self.tabs.close_current_tab();
        }
        if pressed(&SAVE_TABS) {
            match self.tabs.save() {
                Ok(()) => self.info("Onglets enregistrés"),
                Err(e) => self.error(format!("Enregistrement impossible : {e}")),
            }
        }
        if pressed(&REFRESH) {
            self.refresh_schemas();
        }
        if self.worker.is_running() && ctx.input_mut(|i| i.consume_key(Modifiers::NONE, Key::Escape)) {
            self.cancel();
        }
    }

    fn on_close(&mut self) {
        if let Err(e) = self.tabs.save() {
            tracing::error!("saving tabs: {e}");
        }
        self.config.theme = self.theme.to_config();
        let _ = self.config.save();
        self.updater.schedule_on_quit();
        *self.exit_action.lock().unwrap() = self.updater.exit_action.clone();
    }
}

impl eframe::App for App {
    fn update(&mut self, ctx: &egui::Context, _frame: &mut eframe::Frame) {
        for ev in self.worker.poll() {
            self.handle_event(ev);
        }
        super::status::poll_updater(self);
        self.shortcuts(ctx);

        super::status::update_banner(self, ctx);
        super::status::status_bar(self, ctx);
        egui::SidePanel::left("sidebar")
            .resizable(true)
            .default_width(260.0)
            .width_range(180.0..=500.0)
            .show(ctx, |ui| super::sidebar::show(self, ui));
        egui::TopBottomPanel::bottom("results")
            .resizable(true)
            .default_height(320.0)
            .height_range(120.0..=f32::INFINITY)
            .show(ctx, |ui| super::results::show(self, ui));
        egui::CentralPanel::default().show(ctx, |ui| super::editor::show(self, ui));
        super::dialogs::show(self, ctx);

        if self.busy.is_some() {
            ctx.request_repaint_after(std::time::Duration::from_millis(100));
        }
        if ctx.input(|i| i.viewport().close_requested()) {
            self.on_close();
        }
    }
}

pub fn poll_updater(app: &mut App) {
    while let Some(ev) = app.updater.poll() {
        match ev {
            UpdateEvent::UpToDate => app.info("storingUnicorns est à jour"),
            UpdateEvent::Available(_) => {}
            UpdateEvent::Ready => app.success("Mise à jour installée : redémarrez pour l'appliquer"),
            UpdateEvent::Error(e) => app.error(format!("Mise à jour : {e}")),
        }
    }
}

/// Thin banner above everything while an update is available / downloading / ready.
pub fn update_banner(app: &mut App, ctx: &egui::Context) {
    let state = app.updater.state.clone();
    let show = match &state {
        UpdateState::Available(_) => !app.updater.dismissed,
        UpdateState::Downloading(_) | UpdateState::Ready(_) => true,
        _ => false,
    };
    if !show {
        return;
    }
    egui::TopBottomPanel::top("update").show(ctx, |ui| {
        ui.horizontal(|ui| {
            match &state {
                UpdateState::Available(info) => {
                    ui.label(RichText::new(icon::ARROW_CIRCLE_UP).color(theme::ACCENT));
                    ui.label(format!("La version {} est disponible.", info.version));
                    if ui.button("Installer").clicked() {
                        app.updater.install();
                    }
                    ui.hyperlink_to("Nouveautés", &info.page_url);
                    if ui.button("Plus tard").clicked() {
                        app.updater.dismissed = true;
                    }
                    if ui.button("Ignorer cette version").clicked() {
                        app.config.skipped_version = Some(info.version.to_string());
                        app.updater.dismissed = true;
                        app.save_config();
                    }
                }
                UpdateState::Downloading(info) => {
                    ui.spinner();
                    ui.label(format!("Téléchargement de la version {}…", info.version));
                }
                UpdateState::Ready(info) => {
                    ui.label(RichText::new(icon::CHECK_CIRCLE).color(SUCCESS));
                    ui.label(format!("Version {} prête.", info.version));
                    if ui.button("Redémarrer maintenant").clicked() {
                        app.updater.schedule_restart();
                        ui.ctx().send_viewport_cmd(egui::ViewportCommand::Close);
                    }
                }
                _ => {}
            }
        });
    });
}

pub fn status_bar(app: &mut App, ctx: &egui::Context) {
    egui::TopBottomPanel::bottom("status").show(ctx, |ui| {
        ui.horizontal(|ui| {

                Some(s) => {

                    ui.label(RichText::new(icon::CIRCLE).weak());
                    ui.label(RichText::new("Déconnecté").weak());

                ui.label(format!("{text} {:.1} s", since.elapsed().as_secs_f32()));
                if app.worker.is_running() && ui.small_button(format!("{} Annuler (Échap)", icon::STOP)).clicked() {
                    app.cancel();
                }
            } else if let Some((done, total, label)) = &app.progress {

                ui.label(format!("{done}/{total} {label}"));

                let color = match app.status.kind {
                    StatusKind::Info => ui.visuals().text_color(),
                    StatusKind::Success => SUCCESS,

                };
                ui.label(RichText::new(&app.status.text).color(color));

            ui.with_layout(egui::Layout::right_to_left(egui::Align::Center), |ui| {

                    app.theme = app.theme.next();
                    theme::apply(ui.ctx(), app.theme);
                    app.config.theme = app.theme.to_config();
                    app.save_config();
                }
                let version = format!("v{}", crate::updater::current_version());
                if ui.link(RichText::new(version).weak()).on_hover_text("Vérifier les mises à jour").clicked() {
                    app.updater.check(true, None);
                }
            });
        });
    });
}

fn default_port(t: &DatabaseType) -> &'static str {
    match t {
        DatabaseType::Postgres => "5432",
        DatabaseType::MySQL => "3306",
        DatabaseType::SQLite => "",
        DatabaseType::SQLServer | DatabaseType::Azure => "1433",

pub struct ConnectionForm {

    pub editing: Option<usize>,

    pub name: String,
    pub db_type: DatabaseType,
    pub host: String,

impl ConnectionForm {
    pub fn new(edit: Option<(usize, ConnectionConfig)>) -> Self {
        match edit {
            Some((idx, c)) => Self {
                editing: Some(idx),
                original_name: Some(c.name.clone()),
                port: c.port.map(|p| p.to_string()).unwrap_or_default(),
                host: c.host.unwrap_or_default(),
                username: c.username.unwrap_or_default(),
                password: c.password.unwrap_or_default(),
                azure_auth: c.azure_auth_method.unwrap_or_default(),
                tenant_id: c.tenant_id.unwrap_or_default(),
                name: c.name,
                db_type: c.db_type,
                database: c.database,
                error: None,
                testing: false,
                test_result: None,
            },
            None => {
                let d = ConnectionConfig::default();
                Self::new(Some((usize::MAX, ConnectionConfig { name: String::new(), ..d }))).fresh()
            }
        }
    }

    fn fresh(mut self) -> Self {
        self.editing = None;
        self.original_name = None;
        self
    }

    pub fn set_db_type(&mut self, t: DatabaseType) {
        if self.port.is_empty() || self.port == default_port(&self.db_type) {
            self.port = default_port(&t).to_string();
        }
        self.db_type = t;
    }

    fn uses_server(&self) -> bool {
        self.db_type != DatabaseType::SQLite
    }

    /// Build the config, or a French error message for the form.
    pub fn validate(&self, existing: &[ConnectionConfig]) -> Result<ConnectionConfig, String> {
        let name = self.name.trim();
        if name.is_empty() {
            return Err("Le nom est obligatoire".into());
        }
        if self.original_name.as_deref() != Some(name) && existing.iter().any(|c| c.name == name) {
            return Err("Une connexion porte déjà ce nom".into());
        }
        if self.database.trim().is_empty() {
            return Err(if self.uses_server() { "La base est obligatoire" } else { "Le fichier est obligatoire" }.into());
        }
        let port = if self.uses_server() && !self.port.trim().is_empty() {
            Some(self.port.trim().parse::<u16>().map_err(|_| "Le port doit être un nombre".to_string())?)
        } else {
            None
        };
        let opt = |s: &str| (!s.trim().is_empty()).then(|| s.trim().to_string());
        let server = self.uses_server();
        Ok(ConnectionConfig {
            name: name.to_string(),
            db_type: self.db_type.clone(),
            host: if server { opt(&self.host) } else { None },
            port,
            username: if server { opt(&self.username) } else { None },
            password: if server { (!self.password.is_empty()).then(|| self.password.clone()) } else { None },
            database: self.database.trim().to_string(),
            azure_auth_method: (self.db_type == DatabaseType::Azure).then(|| self.azure_auth.clone()),
            tenant_id: if self.db_type == DatabaseType::Azure { opt(&self.tenant_id) } else { None },
        })
    }

        let mut action = FormAction::None;
        ui.heading(if self.editing.is_some() { "Modifier la connexion" } else { "Nouvelle connexion" });
        ui.add_space(6.0);

            ui.label("Nom");
            ui.text_edit_singleline(&mut self.name);
            ui.end_row();

            ui.label("Type");
            let mut t = self.db_type.clone();
            egui::ComboBox::from_id_salt("db_type")
                .selected_text(t.to_string())
                .show_ui(ui, |ui| {
                    for d in DB_TYPES {
                        let label = d.to_string();
                        ui.selectable_value(&mut t, d, label);
                    }
                });
            if t != self.db_type {
                self.set_db_type(t);
            }
            ui.end_row();

            if self.uses_server() {
                ui.label("Hôte");
                ui.text_edit_singleline(&mut self.host);
                ui.end_row();
                ui.label("Port");
                ui.add(egui::TextEdit::singleline(&mut self.port).desired_width(80.0));
                ui.end_row();
                if self.db_type == DatabaseType::Azure {
                    ui.label("Authentification");
                    egui::ComboBox::from_id_salt("azure_auth")
                        .selected_text(self.azure_auth.to_string())
                        .show_ui(ui, |ui| {
                            for m in AZURE_METHODS {
                                let label = m.to_string();
                                ui.selectable_value(&mut self.azure_auth, m, label);
                            }

        .collect()
}

    match t {
        SqlToken::Keyword(s)
        | SqlToken::Function(s)
        | SqlToken::String(s)
        | SqlToken::Number(s)
        | SqlToken::Operator(s)
        | SqlToken::Comment(s)
        | SqlToken::Identifier(s)
        | SqlToken::Column(s)
        | SqlToken::Punctuation(s)
        | SqlToken::Whitespace(s) => s,
    }
}

pub fn highlight_job(text: &str, known_columns: &[String], dark: bool, size: f32) -> LayoutJob {
    let mut job = LayoutJob::default();
    let font = FontId::monospace(size);
    let mut covered = 0;
    for token in tokenize_sql(text, known_columns) {
        let s = token_text(&token);
        covered += s.len();
        job.append(s, 0.0, TextFormat { font_id: font.clone(), color: token_color(&token, dark), ..Default::default() });
    }
    // The tokenizer must never drop text; fall back to plain text if it does.
    if covered != text.len() {
        job = LayoutJob::single_section(text.to_string(), TextFormat { font_id: font, ..Default::default() });
    }
    job
}

/// Replace the token ending at `cursor` with `suggestion`; returns the new
/// text and cursor (bytes).
pub fn apply_suggestion(text: &str, cursor: usize, suggestion: &str) -> (String, usize) {
    let cursor = cursor.min(text.len());
    let (start, _, _) = extract_token(&text[..cursor]);
    let mut out = String::with_capacity(text.len() + suggestion.len());
    out.push_str(&text[..start]);
    out.push_str(suggestion);
    out.push_str(&text[cursor..]);
    (out, start + suggestion.len())
}

    let trigger = ui.input_mut(|i| i.consume_key(Modifiers::COMMAND, Key::Space));
    if !app.completion.open {
        return trigger;
    }
    let n = app.completion.items.len();
    ui.input_mut(|i| {
        if i.consume_key(Modifiers::NONE, Key::ArrowDown) {
            app.completion.selected = (app.completion.selected + 1) % n;
        }
        if i.consume_key(Modifiers::NONE, Key::ArrowUp) {
            app.completion.selected = (app.completion.selected + n - 1) % n;
        }
        if i.consume_key(Modifiers::NONE, Key::Escape) {
            app.completion.open = false;
        }
    });
    let accept = ui.input_mut(|i| i.consume_key(Modifiers::NONE, Key::Enter) || i.consume_key(Modifiers::NONE, Key::Tab));
    if accept {
        accept_selected(app, ui.ctx());
    }
    trigger
}

fn accept_selected(app: &mut App, ctx: &egui::Context) {
    let Some(item) = app.completion.items.get(app.completion.selected).cloned() else { return };
    let tab = app.tabs.current_tab_mut();
    let (text, cursor) = apply_suggestion(&tab.query, app.cursor, &item);
    tab.query = text;
    app.cursor = cursor;
    set_cursor(ctx, &app.tabs.current_tab().query, cursor);
    app.completion.open = false;
}

/// After the TextEdit: (re)compute suggestions and draw the popup under the cursor.
pub fn after_edit(app: &mut App, ui: &mut egui::Ui, out: &egui::text_edit::TextEditOutput, triggered: bool) {
    let typed_word_char = ui.input(|i| {
        i.events.iter().any(|e| matches!(e, egui::Event::Text(t) if t.chars().all(|c| c.is_alphanumeric() || c == '_' || c == '.')))
    });
    if triggered || (out.response.changed() && typed_word_char) {
        refresh(app);
    } else if out.response.changed() || !out.response.has_focus() {
        app.completion.open = false;
    }
    if !app.completion.open {
        return;
    }
    let Some(range) = out.cursor_range else { return };
    let rect = out.galley.pos_from_cursor(&range.primary);
    let pos = out.galley_pos + rect.left_bottom().to_vec2();
    let mut clicked = None;
    egui::Area::new(egui::Id::new("completion"))
        .order(egui::Order::Foreground)

    pub fn new(kind: BatchKind, schemas: &[SchemaInfo]) -> Self {
        let items = schemas

            .flat_map(|s| s.tables.iter().map(move |t| (s.name.clone(), t.clone(), false)))
            .collect();
        Self { kind, items, dir: None, format: ExportFormat::Csv }

        self.items.iter_mut().for_each(|i| i.2 = on);

    fn select_matching_files(&mut self) {

        }
    }

pub fn format_for(path: &Path) -> ExportFormat {
    match path.extension().and_then(|e| e.to_str()).map(|e| e.to_ascii_lowercase()) {
        Some(e) if e == "sql" => ExportFormat::SqlInsert,
        _ => ExportFormat::Csv,
    }
}

/// Export the current result: native save dialog, then a background write.
pub fn export_result(app: &mut App) {
    let Some(view) = &app.results else { return };
    let table = view.table.clone().unwrap_or_else(|| "table".into());
    let file_name = format!("{}.csv", BatchExportState::clean_table_name(&table));
    let Some(path) = rfd::FileDialog::new()
        .add_filter("CSV", &["csv"])
        .add_filter("SQL INSERT", &["sql"])
        .set_file_name(file_name)
        .save_file()
    else {
        return;
    };
    let format = format_for(&path);
    app.worker.export_result(view.result.clone(), format, path, table, app.quotes());
